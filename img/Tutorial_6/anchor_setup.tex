% Drag anchor (Tutorial 6), in the style of cube_setup.png, cpt_setup.png and
% sphere_setup.png, at the start of stage 2. A schematic of the half model: the soil
% block (100 x 10 x 18.1 m, of 3.2 m base elements) is drawn shortened, with 5 x 2 x 2
% elements. Only the window of soil between the two boundary-track planes - 0.5 m
% behind and 2.5 m ahead of the anchor - is active, and only it is drawn with GIMP
% domains; the two elements the anchor rests in are refined once (schematic; the
% finest elements are 0.1 m). The anchor, its pull line and the tracked fluke tip are
% at their true size and position relative to one element. Lengths are in units of
% the 3.2 m element.
% anchor_triangles.tex and anchor_gimps.tex are written by anchor_triangles.py.
% Build: python3 anchor_triangles.py path/to/MPM3S/validation_data_and_jsons
%        pdflatex anchor_setup.tex
%        pdftocairo -png -transp -r 300 -singlefile anchor_setup.pdf anchor_setup

\newcommand{\mpoint}[1]{\fill[mpnavy] #1 circle (0.8pt);}
\newcommand{\activenode}[1]{\path[activenode] #1 circle (1.8pt);}
\newcommand{\gridnode}[1]{\fill[black] #1 circle (1.1pt);}
% one GIMP domain, lower corner (#1,#2,#3), side #4: the material point at the
% centre, then the three visible faces
\newcommand{\gimp}[4]{%
  \mpoint{({#1+#4/2},{#2+#4/2},{#3+#4/2})}
  \filldraw[gimpedge, fill=gimpfill, fill opacity=0.4]
    (#1,#2,#3) -- ({#1+#4},#2,#3) -- ({#1+#4},#2,{#3+#4}) -- (#1,#2,{#3+#4}) -- cycle;
  \filldraw[gimpedge, fill=gimpside, fill opacity=0.4]
    ({#1+#4},#2,#3) -- ({#1+#4},{#2+#4},#3) -- ({#1+#4},{#2+#4},{#3+#4}) -- ({#1+#4},#2,{#3+#4}) -- cycle;
  \filldraw[gimpedge, fill=gimptop, fill opacity=0.4]
    (#1,#2,{#3+#4}) -- ({#1+#4},#2,{#3+#4}) -- ({#1+#4},{#2+#4},{#3+#4}) -- (#1,{#2+#4},{#3+#4}) -- cycle;}
