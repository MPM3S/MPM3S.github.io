% Plough (Tutorial 5), in the style of anchor_setup.png, at the start of stage 2. A schematic of the
% half model: the soil block (100 x 20 x 20 m, of 3.2 m base elements) is drawn shortened, with
% 6 x 2 x 2 elements. Only the window of soil between the two boundary-track planes - 5 m behind and
% 10 m ahead of the plough, rounded outward to whole base elements - is active, and only it is drawn
% with GIMP domains; the three elements the plough rests in are refined once (schematic; the finest
% elements are 0.05 m). The grid is schematic: each drawn element is 6.4 m, two of the real 3.2 m base
% elements. The plough, the track planes and the share tip's path are at their true size and position
% on that 6.4 m scale. Lengths are in units of the 6.4 m drawn element.
% plough_triangles.tex, plough_points.tex and plough_gimps.tex are written by plough_triangles.py.
% Build: python3 plough_triangles.py path/to/MPM3S/validation_data_and_jsons
%        pdflatex plough_setup.tex
%        pdftocairo -png -transp -r 300 -singlefile plough_setup.pdf plough_setup
\documentclass[border=4pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, calc}

\definecolor{gridgrey}{RGB}{185,185,185}
\definecolor{activeorange}{RGB}{245,130,32}
\definecolor{mpnavy}{RGB}{0,0,153}
\definecolor{gimpfill}{RGB}{179,179,226}
\definecolor{gimpside}{RGB}{158,158,212}
\definecolor{gimptop}{RGB}{204,204,238}
\definecolor{dimpurple}{RGB}{104,36,109}
\definecolor{rigidred}{RGB}{237,28,36}

\tikzset{
  mesh/.style={draw=gridgrey, line width=0.6pt},
  hidden/.style={draw=gridgrey, line width=0.45pt, dash pattern=on 2pt off 1.5pt},
  gimpedge/.style={draw=mpnavy, line width=0.3pt, line join=round},
  activenode/.style={fill=activeorange, draw=black, line width=0.35pt},
  track/.style={draw=dimpurple, line width=0.9pt, dash pattern=on 4pt off 2pt},
  path/.style={draw=rigidred, line width=1.1pt, dash pattern=on 3pt off 1.5pt},
  dimline/.style={draw=dimpurple, line width=1pt},
  dim/.style={dimline, {Latex[length=5pt]}-{Latex[length=5pt]}},
  wall/.style={draw=black, line width=0.8pt},
  hatch/.style={draw=black, line width=0.4pt},
  leader/.style={draw=black, line width=0.4pt},
  label/.style={font=\sffamily\small},
}

\newcommand{\mpoint}[1]{\fill[mpnavy] #1 circle (0.8pt);}
\newcommand{\activenode}[1]{\path[activenode] #1 circle (1.8pt);}
\newcommand{\gridnode}[1]{\fill[black] #1 circle (1.1pt);}
% one GIMP domain, lower corner (#1,#2,#3), side #4: the material point at the
% centre, then the three visible faces
\newcommand{\gimp}[4]{%
  \mpoint{({#1+#4/2},{#2+#4/2},{#3+#4/2})}
  \filldraw[gimpedge, fill=gimpfill, fill opacity=0.4]
    (#1,#2,#3) -- ({#1+#4},#2,#3) -- ({#1+#4},#2,{#3+#4}) -- (#1,#2,{#3+#4}) -- cycle;
  \filldraw[gimpedge, fill=gimpside, fill opacity=0.4]
    ({#1+#4},#2,#3) -- ({#1+#4},{#2+#4},#3) -- ({#1+#4},{#2+#4},{#3+#4}) -- ({#1+#4},#2,{#3+#4}) -- cycle;
  \filldraw[gimpedge, fill=gimptop, fill opacity=0.4]
    (#1,#2,{#3+#4}) -- ({#1+#4},#2,{#3+#4}) -- ({#1+#4},{#2+#4},{#3+#4}) -- (#1,{#2+#4},{#3+#4}) -- cycle;}

\newsavebox{\legmp}\sbox{\legmp}{\tikz[baseline=-0.6ex]\fill[mpnavy] (0,0) circle (1.4pt);}%

\newsavebox{\legnode}\sbox{\legnode}{\tikz[baseline=-0.6ex]\path[activenode] (0,0) circle (2.4pt);}%

\begin{tikzpicture}[x={(2.4cm,0cm)}, y={(1.2cm,0.96cm)}, z={(0cm,2.4cm)}, font=\sffamily]
  \input{plough_points.tex}   % the share tip, its path and the track planes, from plough_triangles.py

  % the inactive soil ahead of the leading plane: faint faces, mesh and grid nodes only
  \fill[gimpfill!30] (4,0,0) -- (6,0,0) -- (6,0,2) -- (4,0,2) -- cycle;
  \fill[gimptop!30] (4,0,2) -- (6,0,2) -- (6,2,2) -- (4,2,2) -- cycle;
  \fill[gimpside!30] (6,0,0) -- (6,2,0) -- (6,2,2) -- (6,0,2) -- cycle;
  % its hidden element lines, inside the soil and on its back and base
  \foreach \y in {1,2} {\foreach \z in {0,1} {\draw[hidden] (4,\y,\z) -- (6,\y,\z);}}
  \foreach \z in {0,1} {\draw[hidden] (5,0,\z) -- (5,2,\z);}
  \foreach \y in {1,2} {\draw[hidden] (5,\y,0) -- (5,\y,2);}

  % the hidden element lines and active nodes of the window: inside the soil, on its back,
  % base and left-hand faces, and inside the three refined elements
  \foreach \y in {1,2} {\foreach \z in {0,1} {\draw[hidden] (0,\y,\z) -- (4,\y,\z);}}
  \foreach \x in {0,...,4} {\foreach \z in {0,1} {\draw[hidden] (\x,0,\z) -- (\x,2,\z);}}
  \foreach \x in {0,...,4} {\foreach \y in {1,2} {\draw[hidden] (\x,\y,0) -- (\x,\y,2);}}
  \foreach \y in {0.5,1} {\foreach \z in {1,1.5} {\draw[hidden] (0,\y,\z) -- (3,\y,\z);}}
  \foreach \x in {0,0.5,...,3} {\draw[hidden] (\x,0,1.5) -- (\x,1,1.5);}
  \foreach \x in {0.5,1.5,2.5} {\draw[hidden] (\x,0,1) -- (\x,1,1);}
  \foreach \x in {0,0.5,...,3} {\foreach \y in {0.5,1} {\draw[hidden] (\x,\y,1) -- (\x,\y,2);}}
  \foreach \x in {0,...,4} {\foreach \y in {1,2} {\foreach \z in {0,1} {\path[activenode, opacity=0.45] (\x,\y,\z) circle (1.8pt);}}}
  \foreach \x in {0,0.5,...,3} {\foreach \z in {1,1.5} {\path[activenode, opacity=0.45] (\x,0.5,\z) circle (1.8pt);}
                                \path[activenode, opacity=0.45] (\x,1,1.5) circle (1.8pt);}
  \foreach \x in {0.5,1.5,2.5} {\path[activenode, opacity=0.45] (\x,1,1) circle (1.8pt);}

  % the GIMP domains of the active window, far to near
  \input{plough_gimps.tex}

  % the visible mesh, with the three refined elements under the plough
  \foreach \x in {0,...,6} {\draw[mesh] (\x,0,0) -- (\x,0,2) -- (\x,2,2);}
  \foreach \z in {0,1,2} {\draw[mesh] (0,0,\z) -- (6,0,\z) -- (6,2,\z);}
  \foreach \y in {1,2} {\draw[mesh] (0,\y,2) -- (6,\y,2) -- (6,\y,0);}
  \foreach \x in {0.5,1.5,2.5} {\draw[mesh] (\x,0,1) -- (\x,0,2) -- (\x,1,2);}
  \draw[mesh] (0,0,1.5) -- (3,0,1.5);
  \draw[mesh] (0,0.5,2) -- (3,0.5,2);

  % the fixed base
  \draw[wall] (0,0,0) -- (6,0,0);
  \foreach \t in {0.01,0.0225,...,1.0} {\draw[hatch] ($(0,0,0)!\t!(6,0,0)$) -- ++(-0.1cm,-0.14cm);}

  % nodes: active in the window, grid nodes in the inactive soil
  \foreach \x in {0,...,4} {\foreach \z in {0,1,2} {\activenode{(\x,0,\z)}} \foreach \y in {1,2} {\activenode{(\x,\y,2)}}}
  \foreach \x in {0.5,1.5,2.5} {\foreach \z in {1,1.5,2} {\activenode{(\x,0,\z)}} \activenode{(\x,1,2)}}
  \foreach \x in {0,...,3} {\activenode{(\x,0,1.5)}}
  \foreach \x in {0,0.5,...,3} {\activenode{(\x,0.5,2)}}
  \foreach \x in {5,6} {\foreach \z in {0,1,2} {\gridnode{(\x,0,\z)}} \foreach \y in {1,2} {\gridnode{(\x,\y,2)}}}
  \foreach \y in {1,2} {\foreach \z in {0,1} {\gridnode{(6,\y,\z)}}}
  % the hidden grid nodes of the inactive soil, inside the soil and on its back and base
  \foreach \y in {1,2} {\foreach \z in {0,1} {\fill[black, opacity=0.35] (5,\y,\z) circle (1.1pt);}}

  % the boundary-track planes: 5 m behind and 10 m ahead of the plough, on whole base elements
  \foreach \x in {\trailx,\leadx} {\draw[track] (\x,0,0) -- (\x,0,2) -- (\x,2,2) -- (\x,2,0) -- cycle;}

  % the share tip's path: stage 2 down to 1.85 m over two plough lengths, then level for 25 m in stage 3
  % (drawn shortened, to the end of the block)
  \draw[path] (tip) -- (end2);
  \draw[path, -{Latex[length=5pt, width=4pt]}] (end2) -- (5.85,0,\pathz);

  % the plough
  \input{plough_triangles.tex}
  \fill[rigidred] (tip) circle (1.6pt);
  \draw[dimline, -{Latex[length=6pt, width=5pt]}, line width=1.3pt] ($(bladelabel)+(0.9cm,0.45cm)$) -- ++(0.9cm,0) node[right=1pt, dimpurple, label, fill=white, inner sep=1.5pt] {$0.2$ m/s};

  % labels
  \draw[leader] (bladelabel) -- ++(-0.5cm,1.2cm) node[above, label] {plough};
  \draw[leader] (tip) -- ++(1.3cm,0.35cm) node[right, label, fill=white, inner sep=1.5pt] {share tip};
  \draw[leader] ($(tip)!0.5!(end2)$) -- ++(0.4cm,-1.2cm) node[below, label, fill=white, inner sep=1.5pt, align=center] {stage 2: down $1.85$ m\\over $2L$};
  \draw[leader] ($(end2)!0.45!(5.85,0,\pathz)$) -- ++(0.2cm,-0.9cm) node[below, label, fill=white, inner sep=1.5pt, align=center] {stage 3: level\\for $25$ m};
  \draw[leader] (\trailx,0,0.5) -- ++(-1.2cm,0) node[left, label, align=right] {trailing\\plane};
  \draw[leader] (\leadx,0,0.5) -- ++(1.1cm,-0.64cm) node[right, label, align=left] {leading\\plane};
  \node[label, gray, align=center] at (5.0,1.0,2.0) {inactive soil, switched on\\as the leading plane advances};

  % gravity
  \draw[dimline, -{Latex[length=6pt, width=5pt]}, line width=1.3pt] (-0.55,0,2.0) -- ++(0,-1.0cm) node[below, dimpurple, label] {$g$};

  % dimensions

  \draw[dim] ($(6,0,0)+(0.4cm,-0.3cm)$) -- node[below right=0pt] {20 m} ($(6,2,0)+(0.4cm,-0.3cm)$);
